result<std::uint64_t> described_mip_size(const dxgi_format_descriptor& descriptor, std::uint32_t width,

  const std::uint64_t blocks_wide =
      descriptor.block_width == 1U ? width : std::max<std::uint32_t>(1U, (width + 3U) / 4U);
